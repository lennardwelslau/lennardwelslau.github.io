%---------------------------------------------------------------- entries
% One full-width row: bold/italic label on the left, dates flush right.
% The row fills \textwidth, so whatever follows starts on the next line.
\newcommand{\cvrow}[2]{%
  \begin{tabular*}{\textwidth}[t]{@{}l@{\extracolsep{\fill}}r@{}}
    #1 & {\footnotesize #2} \\
  \end{tabular*}
}

% \cventry{institution}{, role}{dates}{body} - a standalone position
\newcommand{\cventry}[4]{\cvrow{\textbf{#1}\textit{\small #2}}{#3}
  {\small #4}
  \vspace{\entrygap}}

% \cvhead{institution} - institution heading with several roles beneath.
% This deliberately does not use \cvrow: \cvrow leaves a trailing space so
% that body text starts on the next line, and combined with the blank line
% before the first role that opened a paragraph and left the institution
% floating well above the role belonging to it. Closing the paragraph here
% instead puts the first role the same distance below its heading as a
% \cventry body sits below its own.
\newcommand{\cvhead}[1]{%
  \begin{tabular*}{\textwidth}[t]{@{}l@{\extracolsep{\fill}}r@{}}
    \textbf{#1} & \\
  \end{tabular*}%
  \par\nobreak
}

% \cvrole{role}{dates}{body} - a role under \cvhead
\newcommand{\cvrole}[3]{\cvrow{\textit{\small #1}}{#2}
  {\small #3}}

% \cvrolelast{role}{dates}{body} - the final role under \cvhead
\newcommand{\cvrolelast}[3]{\vspace{\rolegap}\cvrole{#1}{#2}{#3}\vspace{\entrygap}}

% \myitem{text} - dash-led bullet with hanging indent
\newcommand{\myitem}[1]{\hangindent=1em\hangafter=1\textbf{--}~#1\par}
